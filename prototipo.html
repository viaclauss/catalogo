/* 1. CONFIGURAÇÕES PRINCIPAIS */ /* Aqui ficam os dados vitais da loja. Se o número mudar ou a planilha mudar, é só alterar aqui. */ const CONFIG = { nomeLoja: "Via Clauss", whatsapp: "5543998155361", sheetCsvUrl: "https://docs.google.com/..." /* O link da sua planilha do Google */ }; /* Variáveis que vão guardar o estado da loja enquanto a pessoa navega */ let produtos = []; /* Vai guardar a lista de produtos que vier da planilha */ let carrinho = []; /* Vai guardar o que o cliente colocar na sacola */ let categoriaAtiva = ""; /* Diz qual aba (Roupas, Semijoias, etc) está selecionada no momento */ /* 2. BUSCANDO OS PRODUTOS */ /* O 'async' significa que o site vai fazer isso em segundo plano, sem travar a tela. */ async function carregarProdutos() { try { /* Tenta ir lá no link do Google Sheets baixar o texto (CSV) */ const res = await fetch(CONFIG.sheetCsvUrl); if (!res.ok) throw new Error("Não foi possível carregar a planilha."); /* Pega o texto da planilha e manda para a função 'parseCsv' para transformar em lista */ produtos = parseCsv(await res.text()); /* Depois que os produtos chegaram, ele cria os botões de filtro e desenha as roupas na tela */ popularFiltros(); renderizar(); } catch (e) { /* Se a internet cair ou o link der erro, mostra essa mensagem na tela */ document.getElementById("conteudo").innerHTML = `

⚠️ Erro ao carregar produtos.

`; } } /* 3. TRADUZINDO A PLANILHA (CSV para JavaScript) */ function parseCsv(text) { const linhas = text.trim().split("\n"); /* Corta o texto em várias linhas */ const cab = linhas[0].split(","); /* Pega a primeira linha, que é o cabeçalho (Nome, Preço, etc) */ /* Pega do produto 1 em diante e transforma num formato que o site entende (Objetos) */ return linhas.slice(1).map(linha => { const cols = parseLinha(linha); const obj = {}; cab.forEach((key, i) => obj[key] = (cols[i] || "")); /* Tratamento matemático: Transforma os preços que estão escritos "R$ 100,00" no número 100 */ const ps = (obj.preco || "").toString().replace(/[^\d.,]/g, "").trim(); obj.preco = parseFloat(ps) || 0; /* Vê quanto tem no estoque (se estiver vazio na planilha, ele finge que tem 99) */ obj.estoque = obj.estoque !== "" && obj.estoque !== undefined ? parseInt(obj.estoque) : 99; return obj; }).filter(p => p.nome); /* Só guarda na memória se a linha tiver um "nome" preenchido */ } /* 4. DESENHANDO OS PRODUTOS NA TELA */ function renderizar() { const termo = document.getElementById("busca").value.toLowerCase(); /* O que a pessoa digitou na lupa */ const conteudo = document.getElementById("conteudo"); /* Cria uma "sub-lista" apenas com os produtos que batem com a busca ou com a categoria clicada */ const filtrados = produtos.filter(p => { const mb = p.nome.toLowerCase().includes(termo); /* Vê se o nome tem a letra buscada */ if (!mb) return false; if (categoriaAtiva === "") return true; /* Se não tiver filtro, mostra tudo */ return p.categoria?.toLowerCase() === categoriaAtiva.toLowerCase(); }); conteudo.innerHTML = ""; /* Limpa a tela antes de desenhar */ /* Manda cada produto filtrado para a função 'criarCard' e joga na tela */ filtrados.forEach(prod => { conteudo.appendChild(criarCard(prod)); }); } /* 5. MONTANDO O VISUAL DE CADA PRODUTO */ function criarCard(prod) { const card = document.createElement("div"); card.className = "card"; /* Se não tiver foto na planilha, ele coloca um emoji de casaco (🧥) */ const imgHtml = prod.imagem ? `` : `🧥`; /* Se o estoque for zero ou menor, a variável fica verdadeira */ const semEstoque = prod.estoque <= 0; /* Aqui ele junta a foto, o nome, o preço e cria o botão */ card.innerHTML = `
${imgHtml}

${prod.nome}

R$ ${prod.preco.toFixed(2)}

`; /* Ensina o botão "Adicionar": quando for clicado, executa a função adicionarAoCarrinho */ if (!semEstoque) { card.querySelector(".btn-add").addEventListener("click", e => adicionarAoCarrinho(prod)); } return card; } /* 6. COLOCANDO NO CARRINHO */ function adicionarAoCarrinho(prod) { /* Verifica se o produto já está no carrinho */ const ex = carrinho.find(i => i.nome === prod.nome); if (ex) { /* Se já tiver, só aumenta a quantidade (+1), respeitando o limite do estoque */ if (ex.qtd >= prod.estoque) return; ex.qtd++; } else { /* Se for novo, coloca na lista com quantidade = 1 */ carrinho.push({ ...prod, qtd: 1 }); } atualizarCarrinho(); /* Atualiza o preço total e o número na bolinha lá em cima */ abrirCarrinho(); /* Puxa o painel lateral pra pessoa ver o que colocou */ } /* 7. FECHANDO A VENDA (O CHECKOUT) */ document.getElementById("btn-enviar-pedido").addEventListener("click", () => { /* Pega os dados que o cliente digitou no formulário final */ const nome = document.getElementById("campo-nome").value; const fone = document.getElementById("campo-fone").value; const obs = document.getElementById("campo-obs").value; /* Monta uma lista bonitinha em formato de texto para o WhatsApp */ const itens = carrinho.map(i => `• ${i.nome} (x${i.qtd}) — R$ ${i.preco}`).join("\n"); const total = carrinho.reduce((s, i) => s + i.preco * i.qtd, 0); /* Escreve a mensagem final que vai chegar no seu celular */ let msg = `*Novo pedido — ${CONFIG.nomeLoja}*\n\n*Cliente:* ${nome}\n*Contato:* ${fone}\n\n*Itens:*\n${itens}\n\n*Total: R$ ${total}*`; /* Abre o WhatsApp da loja (o link wa.me) já com o texto preenchido! */ window.open(`https://wa.me/${CONFIG.whatsapp}?text=${encodeURIComponent(msg)}`, "_blank"); });
